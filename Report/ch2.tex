\section{DNA Designing}
DNA constructs are designed through the custom software engine:
\begin{itemize}
    \item \textbf{Construct Sizing}: Standardized to a 200~base pair (bp) double-stranded DNA construct.
    \item \textbf{Structure}: Flanked by 20~bp forward and reverse primers, containing four internal 24~bp fluorescent probe targets (FAM, HEX, Texas Red, Cy5) separated by spacers (Figure~\ref{fig:molecular_anatomy}).
    \item \textbf{Thermodynamic Boundaries}: Melting temperatures ($T_m$) are calculated using SantaLucia parameters:
    \begin{equation}
        T_m = \frac{\Delta H^\circ}{\Delta S^\circ + R \ln(C_T / 4)} + 16.6 \log_{10}[\text{Na}^+]
    \end{equation}
    Primers are matched at $T_m = 58.5 \pm 1.0^\circ\text{C}$, and probes are set at $T_m \ge 68.0^\circ\text{C}$ ($+10.0^\circ\text{C}$ above primers).
    \item \textbf{Safety Screening}: Local BLAST checks ensure zero homology ($<25\%$) with any known environmental or human pathogens.
\end{itemize}

\section{Physical Synthesis \& Formulation}
\begin{enumerate}
    \item \textbf{Plasmid Miniprep}: Synthesized in pUC57 vector (Lot No. BJ033535).
    \item \textbf{PCR Scaling}: High-yield PCR amplification ($35\text{ cycles}$) produces pure 200~bp amplicon.
    \item \textbf{1.0-Liter Bulk Formulation}: 100~$\mu\text{L}$ PCR product is mixed into 1.0~Liter of stabilized carrier buffer (10~mL 1M Tris-HCl, 2~mL 0.5M EDTA, 988~mL $\text{ddH}_2\text{O}$), generating the master taggant stock.
\end{enumerate}

\section{Product Integration}
\begin{itemize}
    \item \textbf{Liquids}: Directly dosed into pharmaceutical bottles (\textit{Gelsemium 30, Surkhana}) at parts-per-billion levels.
    \item \textbf{Packaging \& Labels}: Applied via an automated 6-station roll-to-roll spray machine with infrared flash curing ($60^\circ\text{C}$, 1.5s).
\end{itemize}

\section{Field Recovery}
\begin{enumerate}
    \item \textbf{Swabbing}: A sterile swab moistened with recovery buffer wipes the product security zone.
    \item \textbf{Chemical Etching}: For coated samples, mild fluoride buffer ($\text{NH}_4\text{F}/\text{HF}$) dissolves silica coatings in 10 minutes.
    \item \textbf{Spin Column}: Standard silica spin column elutes clean DNA in $25.0~\mu\text{L}$ water.
\end{enumerate}

\section{qPCR Verification}
\begin{itemize}
    \item \textbf{Multiplex Run}: The recovered sample is run in a 4-channel real-time thermal cycler to simultaneously detect FAM, HEX, Texas Red, and Cy5.
    \item \textbf{Database Matching}: Recorded $C_q$ values are checked against the database to confirm product authenticity and lot identity.
\end{itemize}
